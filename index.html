RGB To Hex Conversion from Codewars


function rgb(r, g, b) {
  const clip = (value) => 
    Math.max(0, Math.min(255, value));
  const r1 = clip(r);
  const g1 = clip(g);
  const b1 = clip(b);
  const toHex = (c) => 
    c.toString(16).padStart(2, '0').toUpperCase();
  return toHex(r1) + toHex(g1) + toHex(b1);
}
